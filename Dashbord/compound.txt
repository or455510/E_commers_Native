function getProductPhotos($product_id){
    global $db;
    $stmt = $db->prepare("SELECT * FROM product_photos WHERE product_id = ?");
    $stmt->execute([$product_id]);
    return $stmt->fetchAll(PDO::FETCH_ASSOC);
}

<?php
require_once 'config/db.php'; // الاتصال بقاعدة البيانات
require_once 'config/functions.php';

if(isset($_POST['upload_photo'])){
    $product_id = $_POST['product_id'];

    if(isset($_FILES['photo']) && $_FILES['photo']['error'] === 0){
        $file_tmp = $_FILES['photo']['tmp_name'];
        $file_name = $_FILES['photo']['name'];
        $file_ext = strtolower(pathinfo($file_name, PATHINFO_EXTENSION));

        // تحقق من نوع الصورة
        $allowed = ['jpg','jpeg','png','gif'];
        if(in_array($file_ext, $allowed)){
            // مسار حفظ الصورة
            $upload_dir = 'uploads/products/';
            if(!is_dir($upload_dir)){
                mkdir($upload_dir, 0777, true);
            }

            // اسم فريد لتجنب تكرار الملفات
            $new_name = uniqid() . '.' . $file_ext;
            $destination = $upload_dir . $new_name;

            if(move_uploaded_file($file_tmp, $destination)){
                // حفظ اسم الصورة في قاعدة البيانات
                global $db;
                $stmt = $db->prepare("INSERT INTO product_photos (product_id, image) VALUES (?, ?)");
                if($stmt->execute([$product_id, $new_name])){
                    echo "تم رفع الصورة بنجاح!";
                } else {
                    echo "حدث خطأ أثناء حفظ الصورة في قاعدة البيانات.";
                }
            } else {
                echo "فشل رفع الصورة على السيرفر.";
            }
        } else {
            echo "صيغة الملف غير مسموح بها.";
        }
    } else {
        echo "لم يتم اختيار أي صورة أو حدث خطأ أثناء الرفع.";
    }
}
